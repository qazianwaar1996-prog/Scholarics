# Disable Cloudflare's automatic "serve page.html at /page" behavior.
# Without this, every .html file is accessible at BOTH /page.html AND /page,
# creating duplicate content that causes GSC "Alternate page with proper canonical"
# and "Page with redirect" errors (73+ affected pages).
# With html_handling = "none", only the explicit .html URL is served;
# extensionless hits are handled by the redirects in _redirects.
[build]

[placement]

[pages]
html_handling = "none"
